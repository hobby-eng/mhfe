# The vendored C files that scripts/build-argon2-wasm.sh compiles for the browser, the one list of
# them: one file per line as "<part> <path>", the path from the repository root.
# packaging/Dockerfile.reproducible copies this file for that build. Native builds compile the
# copy in mhfe-core. The parts:
#
#   common    the code around the Argon2 core
#   portable  the portable core, as the SIMD core opt.c gave no speed-up in a browser and would
#             exclude browsers without WebAssembly SIMD
#
# The build takes the files in the order of these lines; another order would change its bytes.
common vendor/phc-winner-argon2/src/argon2.c
common vendor/phc-winner-argon2/src/core.c
common vendor/phc-winner-argon2/src/encoding.c
common vendor/phc-winner-argon2/src/thread.c
portable vendor/phc-winner-argon2/src/ref.c
common vendor/phc-winner-argon2/src/blake2/blake2b.c
